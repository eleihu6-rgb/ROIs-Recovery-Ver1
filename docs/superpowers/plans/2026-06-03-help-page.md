# Help Page Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a Help page to the gantt app covering Live, Scenario, Settings & Personalization, R'Bot, and a Glossary — with Playwright-captured screenshots and a smoke E2E test.

**Architecture:** `'help'` is added to `ActiveModule`; a Help tab appears after Regression in the top nav; when Help is active the sidebar auto-collapses to `'hidden'` giving Help the full viewport. The page is a two-column layout: 260 px left nav (search + expandable category tree) and a scrollable content area that shows either a welcome screen or an article (format: overview box → numbered steps → controls reference table).

**Tech Stack:** React 19, Zustand, Tailwind CSS / `@rois/ui` tokens, Lucide icons, Playwright for screenshots.

  ```tsx
  // gantt/src/components/help/help-nav.tsx
  import { useState } from 'react'
  import { Search, CalendarDays, FlaskConical, Settings2, MessageSquare, BookOpen, ChevronDown, ChevronRight } from 'lucide-react'
  import { HELP_CATEGORIES } from './help-data'
  import type { HelpCategory } from './help-data'

  const CATEGORY_ICONS: Record<string, React.ElementType> = {
    CalendarDays,
    FlaskConical,
    Settings2,
    MessageSquare,
    BookOpen,
  }

  interface HelpNavProps {
    activeTopic: string | null
    onTopicSelect: (slug: string) => void
  }

  export const HelpNav = ({ activeTopic, onTopicSelect }: HelpNavProps) => {
    const [query, setQuery] = useState('')
    const [expanded, setExpanded] = useState<Record<string, boolean>>(
      Object.fromEntries(HELP_CATEGORIES.map((c) => [c.slug, c.defaultExpanded])),
    )

    const q = query.trim().toLowerCase()

    const toggle = (slug: string) =>
      setExpanded((prev) => ({ ...prev, [slug]: !prev[slug] }))

    const matchingTopics = (cat: HelpCategory) =>
      q ? cat.topics.filter((t) => t.title.toLowerCase().includes(q)) : cat.topics

    return (
      <div className="flex h-full flex-col">
        {/* Header */}
        <div className="flex-shrink-0 border-b border-border px-4 py-3">
          <p className="text-sm font-semibold text-foreground mb-2">Help Center</p>
          <div className="flex items-center gap-1.5 rounded-md border border-border bg-background px-2 py-1.5">
            <Search className="h-3.5 w-3.5 shrink-0 text-muted-foreground" />
            <input
              type="text"
              value={query}
              onChange={(e) => setQuery(e.target.value)}
              placeholder="Search topics…"
              className="w-full bg-transparent text-xs text-foreground placeholder:text-muted-foreground outline-none"
            />
          </div>
        </div>

        {/* Category tree */}
        <nav className="flex-1 overflow-y-auto py-2">
          {HELP_CATEGORIES.map((cat) => {
            const topics = matchingTopics(cat)
            if (q && topics.length === 0) return null
            const isOpen = q ? true : (expanded[cat.slug] ?? cat.defaultExpanded)
            const Icon = CATEGORY_ICONS[cat.lucideIcon] ?? BookOpen

            return (
              <div key={cat.slug}>
                {/* Category header */}
                <button
                  onClick={() => toggle(cat.slug)}
                  className="flex w-full items-center gap-2 px-4 py-1.5 text-xs font-semibold text-foreground hover:bg-muted/50 transition-colors"
                >
                  <Icon className="h-3.5 w-3.5 shrink-0 text-muted-foreground" />
                  <span className="flex-1 text-left">{cat.title}</span>
                  {isOpen
                    ? <ChevronDown className="h-3 w-3 text-muted-foreground" />
                    : <ChevronRight className="h-3 w-3 text-muted-foreground" />}
                </button>

                {/* Topics */}
                {isOpen && topics.map((topic) => (
                  <button
                    key={topic.slug}
                    onClick={() => onTopicSelect(topic.slug)}
                    className={[
                      'flex w-full items-center px-4 py-1.5 pl-9 text-xs transition-colors text-left',
                      topic.slug === activeTopic
                        ? 'bg-primary/10 text-primary font-medium border-r-2 border-primary'
                        : 'text-muted-foreground hover:text-foreground hover:bg-muted/50',
                    ].join(' ')}
                  >
                    {topic.title}
                  </button>
                ))}
              </div>
            )
          })}
        </nav>
      </div>
    )
  }
  ```

> **Note:** Screenshots reference files in `gantt/public/help/screenshots/`. That directory does not exist yet — the script creates it. Until the script is run, `HelpScreenshot` shows a placeholder div instead of a broken image (handled by the `onError` handler in `help-article.tsx`).

- [ ] **Step 9.1 — Create `capture-help-screenshots.ts`**

  ```ts
  // e2e/scripts/capture-help-screenshots.ts
  /**
   * Run with:
   *   cd e2e && npx ts-node --esm scripts/capture-help-screenshots.ts
   *
   * Prerequisites:
   *   - gantt dev server running on port 5173 (cd gantt && npx vite --port 5173)
   *   - live-server running on port 3000 (required for auth + real data)
   *
   * Output: gantt/public/help/screenshots/*.png (viewport 1280×800)
   */
  import { chromium } from 'playwright'
  import path from 'node:path'
  import fs from 'node:fs/promises'
  import { fileURLToPath } from 'node:url'

  const __dirname   = path.dirname(fileURLToPath(import.meta.url))
  const REPO_ROOT   = path.resolve(__dirname, '../..')
  const OUT_DIR     = path.join(REPO_ROOT, 'gantt/public/help/screenshots')
  const BASE_URL    = process.env.GANTT_BASE_URL ?? 'http://localhost:5173'
  const API_URL     = process.env.GANTT_API_URL  ?? 'http://localhost:3000'
  const USER        = process.env.GANTT_TEST_USER ?? 'admin'
  const PASS        = process.env.GANTT_TEST_PASS ?? '123456'

  await fs.mkdir(OUT_DIR, { recursive: true })

  const browser = await chromium.launch({ headless: true })
  const ctx     = await browser.newContext({ viewport: { width: 1280, height: 800 } })
  const page    = await ctx.newPage()

  // ── Auth helper ───────────────────────────────────────────────────────────────
  async function login() {
    const res = await page.request.post(`${API_URL}/api/auth/login`, {
      data: { userCode: USER, password: PASS },
    })
    const { token } = await res.json() as { token: string }
    await page.addInitScript(
      ({ token, userCode, schema }) => {
        window.sessionStorage.setItem('rois-auth', JSON.stringify({
          user: { userCode, userName: userCode, schema },
          token,
        }))
        const orig = window.fetch
        window.fetch = async (...args) => {
          const init    = args[1] ?? {}
          const headers = new Headers((init as RequestInit).headers ?? {})
          headers.set('Authorization', `Bearer ${token}`)
          args[1] = { ...init, headers } as RequestInit
          return orig.apply(window, args)
        }
      },
      { token, userCode: USER, schema: 'f8' },
    )
  }

  async function save(name: string, el: Parameters<typeof page.locator>[0] | 'page') {
    const dest = path.join(OUT_DIR, `${name}.png`)
    if (el === 'page') {
      await page.screenshot({ path: dest, fullPage: false })
    } else {
      const loc = typeof el === 'string' ? page.locator(el) : el
      await loc.screenshot({ path: dest })
    }
    console.log(`✓  ${name}.png`)
  }

  // ── Navigate to Live ──────────────────────────────────────────────────────────
  await login()
  await page.goto(`${BASE_URL}/fpqe/gantt/`)
  await page.waitForSelector('[data-testid="nav-live"]', { timeout: 30_000 })

  // Trigger Live setup dialog
  await page.click('[data-testid="nav-live"]')
  // Accept the setup dialog defaults and load
  const loadBtn = page.getByRole('button', { name: /load/i })
  if (await loadBtn.isVisible({ timeout: 5_000 }).catch(() => false)) {
    await loadBtn.click()
  }
  // Wait for toolbar to be ready
  await page.waitForSelector('[data-testid="filter-btn"]', { timeout: 30_000 })

  // ── Live toolbar (full toolbar strip) ────────────────────────────────────────
  await save('live-toolbar', '[data-testid="gantt-sub-toolbar"]')

  // ── Date range picker ─────────────────────────────────────────────────────────
  const datePickerBtn = page.locator('[data-testid="date-range-picker"]').first()
    .or(page.locator('button').filter({ hasText: /jan|feb|mar|apr|may|jun|jul|aug|sep|oct|nov|dec/i }).first())
  await datePickerBtn.click()
  await page.waitForTimeout(400)
  await save('live-date-range-open', 'page')
  await page.keyboard.press('Escape')

  // ── Filter button ─────────────────────────────────────────────────────────────
  await save('live-filter-btn', '[data-testid="filter-btn"]')

  // ── Filter dialog — Crew tab ──────────────────────────────────────────────────
  await page.click('[data-testid="filter-btn"]')
  await page.waitForSelector('[role="dialog"]', { timeout: 5_000 })
  // Click the Crew tab (should already be active)
  const crewTab = page.getByRole('tab', { name: /crew/i })
  if (await crewTab.isVisible().catch(() => false)) await crewTab.click()
  // Expand the Rank dropdown
  const rankDropdown = page.getByRole('combobox').filter({ hasText: /rank/i })
    .or(page.locator('button').filter({ hasText: /rank/i })).first()
  if (await rankDropdown.isVisible({ timeout: 2_000 }).catch(() => false)) {
    await rankDropdown.click()
    await page.waitForTimeout(300)
  }
  await save('live-filter-dialog-crew', '[role="dialog"]')
  await page.keyboard.press('Escape') // close rank dropdown if open
  // Apply and capture result
  const applyBtn = page.getByRole('button', { name: /apply/i })
  if (await applyBtn.isVisible({ timeout: 2_000 }).catch(() => false)) await applyBtn.click()
  await page.waitForTimeout(800)
  await save('live-filter-result', '[data-testid="gantt-sub-toolbar"]')
  // Clear filters again
  await page.click('[data-testid="filter-btn"]')
  await page.waitForSelector('[role="dialog"]', { timeout: 5_000 })
  const clearBtn = page.getByRole('button', { name: /clear all/i })
  if (await clearBtn.isVisible({ timeout: 2_000 }).catch(() => false)) await clearBtn.click()
  const applyBtn2 = page.getByRole('button', { name: /apply/i })
  if (await applyBtn2.isVisible({ timeout: 2_000 }).catch(() => false)) await applyBtn2.click()

  // ── Pane buttons ──────────────────────────────────────────────────────────────
  const paneArea = page.locator('[data-testid="reset-layout"]').locator('..')
  await save('live-panes-buttons', paneArea)

  // ── Timezone switcher ─────────────────────────────────────────────────────────
  const tzBtn = page.locator('[data-testid="timezone-switcher"]')
    .or(page.locator('button').filter({ hasText: /utc|asia|america|europe/i }).first())
  if (await tzBtn.isVisible({ timeout: 3_000 }).catch(() => false)) {
    await tzBtn.click()
    await page.waitForTimeout(400)
    await save('live-timezone-open', page.locator('[role="listbox"]').or(page.locator('[role="menu"]')).first())
    await page.keyboard.press('Escape')
  }

  // ── Rule set selector ─────────────────────────────────────────────────────────
  const ruleBtn = page.locator('[data-testid="rule-group-selector"]')
    .or(page.locator('button[aria-label="Rule Set"]').first())
  if (await ruleBtn.isVisible({ timeout: 3_000 }).catch(() => false)) {
    await ruleBtn.click()
    await page.waitForTimeout(400)
    await save('live-ruleset-open', page.locator('[role="listbox"]').or(page.locator('[role="menu"]')).first())
    await page.keyboard.press('Escape')
  }

  // ── Navigate to Scenario ──────────────────────────────────────────────────────
  await page.click('[data-testid="module-nav-scenario"]')
  await page.waitForTimeout(1_000)

  await save('scenario-list', page.locator('[data-testid="scenario-list-panel"]')
    .or(page.locator('.scenario-list-panel')).first())

  // Open create dialog
  const createBtn = page.locator('[data-testid="create-scenario-btn"]')
    .or(page.getByRole('button', { name: /new scenario/i })).first()
  if (await createBtn.isVisible({ timeout: 3_000 }).catch(() => false)) {
    await createBtn.click()
    await page.waitForSelector('[role="dialog"]', { timeout: 5_000 })
    await page.fill('input[placeholder*="name" i], input[placeholder*="scenario" i]', 'June PO — Demo')
    await save('scenario-create-dialog', '[role="dialog"]')
    await page.keyboard.press('Escape')
  }

  // Capture scenario detail states (Draft, Running, Done) if any exist
  const firstScenario = page.locator('[data-testid^="scenario-row-"]').first()
    .or(page.locator('[class*="scenario-row"]').first())
  if (await firstScenario.isVisible({ timeout: 3_000 }).catch(() => false)) {
    await firstScenario.click()
    await page.waitForTimeout(800)
    await save('scenario-detail-draft', page.locator('[data-testid="scenario-detail-panel"]')
      .or(page.locator('[class*="scenario-detail"]')).first())
  }

  // ── Settings / Theme ──────────────────────────────────────────────────────────
  const themeBtn = page.locator('[data-testid="theme-switcher"]')
    .or(page.locator('button[aria-label*="theme" i]').first())
  if (await themeBtn.isVisible({ timeout: 3_000 }).catch(() => false)) {
    await themeBtn.click()
    await page.waitForTimeout(400)
    await save('settings-theme-open', page.locator('[role="menu"]').first())
    await page.keyboard.press('Escape')
  }

  // ── R'Bot ─────────────────────────────────────────────────────────────────────
  const rbotToggle = page.locator('[data-testid="rbot-toggle"]')
    .or(page.locator('button[aria-label*="R\'Bot" i]').first())
    .or(page.locator('button[aria-label*="chat" i]').first())
  if (await rbotToggle.isVisible({ timeout: 3_000 }).catch(() => false)) {
    await rbotToggle.click()
    await page.waitForTimeout(500)
    await save('rbot-panel-open', page.locator('[data-testid="rbot-panel"]')
      .or(page.locator('[class*="ai-chat"]')).first())
    // Type a command and capture response
    const input = page.locator('input[placeholder*="R\'Bot" i], input[placeholder*="Ask" i]').first()
    if (await input.isVisible({ timeout: 2_000 }).catch(() => false)) {
      await input.fill('show only Bangkok crew')
      await input.press('Enter')
      await page.waitForTimeout(3_000)
      await save('rbot-response', page.locator('[data-testid="rbot-panel"]')
        .or(page.locator('[class*="ai-chat"]')).first())
    }
  }

  await browser.close()
  console.log(`\nAll screenshots saved to ${OUT_DIR}`)
  ```
