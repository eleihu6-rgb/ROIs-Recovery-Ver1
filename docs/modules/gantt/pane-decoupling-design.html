2. Layout Configuration Type Definitions

PaneInstanceConfig 

interface PaneInstanceConfig {
  // Unique instance ID (allows multiple of same type)
  instanceId: string
  paneType: PaneType

  // Per-instance viewport state
  viewport: {
    scrollX: number
    scrollY: number
    pxPerHour: number
    dateRange: DateRange | 'sync' // 'sync' = follow global
    leftPanelWidth: number | 'sync'
  }

  // Per-instance interactive state
  selection: {
    frozenRowIds: string[]
    selectedRowIds: string[]
    sortColumn: string | null
    sortDirection: 'asc' | 'desc'
  }

  // Display state
  visible: boolean
  floating: boolean
  floatGeometry?: FloatGeometry
}

LayoutNodeConfig 

type LayoutNodeType = 'split-h' | 'split-v' | 'pane' | 'tab' | 'float'

interface LayoutNodeConfig {
  type: LayoutNodeType
  id: string

  // For split nodes
  children?: LayoutNodeConfig[]
  splitRatio?: number[] // [0.5, 0.5] for equal split

  // For pane/tab nodes
  paneInstanceId?: string
  paneIds?: string[] // For tab container
  activePaneId?: string

  // Size constraints
  minWidth?: number
  minHeight?: number
  flex?: number

  // For float nodes
  geometry?: FloatGeometry
}

RootLayoutConfig (Complete Layout Definition)

interface RootLayoutConfig {
  id: string // Layout preset ID
  name: string // e.g. "Default", "Compare Mode", "Focus Flight"
  root: LayoutNodeConfig // Tree structure of layout
  paneInstances: PaneInstanceConfig[] // All pane instances
  floatingPanes?: LayoutNodeConfig[] // Floating windows

  // Global sync settings (optional sharing)
  syncOptions?: {
    syncScrollX: boolean // Panes can opt-in to sync
    syncZoom: boolean
    syncDateRange: boolean
    syncTimezone: boolean
  }
}
